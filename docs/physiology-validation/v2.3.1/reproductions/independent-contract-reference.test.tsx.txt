it('I9 independent traversal preserves every historical numeric and boolean leaf of all 37 diagnostics',()=>{

 try {

 let numeric=0,boolean=0,missing=0;
 function walk(old:any,now:any,path:string){if(typeof old==='number'||typeof old==='boolean'){if(typeof old==='number')numeric++;else boolean++;expect(now,path).toBe(old);}else if(old&&typeof old==='object'){for(const key of Object.keys(old)){if(now==null||!(key in now))missing++;expect(now!=null&&key in now,path+'.'+key).toBe(true);walk(old[key],now[key],path+'.'+key);}}}
 expect(QA_PROFILES.length).toBe(37);expect(ref.length).toBe(37);
 for(let i=0;i<37;i++){const p=QA_PROFILES[i],d=computeDiagnostic(buildPilotDiagnosticInput(p));expect(ref[i].id).toBe(p.id);walk(ref[i],{id:p.id,effectifs:d.effectifs,limiter:d.limiter,readiness:d.readiness,targets:d.targets,runMLSS:d.runMLSS,synthesis:d.synthesis},p.id);}
 print('I9',{profiles:37,numericLeavesCompared:numeric,booleanLeavesCompared:boolean,missingHistoricalFields:missing});
 }finally{vi.useRealTimers();}
});
